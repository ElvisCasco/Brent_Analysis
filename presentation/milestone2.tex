\begin{frame}{From the previous presentation}
  \footnotesize
  \hd{Original framing.} \emph{How does the topology of the global maritime oil trade network shape the dynamics of oil prices?}
  \begin{itemize}\setlength\itemsep{0.1em}
    \item \hd{Q1.} Can network topology improve oil-price forecasts beyond classical time-series baselines?
    \item \hd{Q2.} How does a maritime disruption propagate into oil prices across horizons, and how long does its effect linger?
  \end{itemize}

  \vspace{0.6em}
  \hd{Why the pivot to causal SCM.}
  \begin{itemize}\setlength\itemsep{0.1em}
    \item Feedback: focus on \hd{causal} estimation rather than forecasting --- forecasting is hard and crowded; a credible counterfactual on a specific disruption is more tractable and policy-relevant.
    \item Reviewed candidate causal methods: event studies, structural VARs (Kilian-style supply/demand decomposition), DiD, network-shock propagation, synthetic control.
    \item \hd{SCM} selected: cleanest combination of \hd{inferential machinery} (in-space placebo, cross-event weight transfer) and \hd{interpretable result} (donor-weighted counterfactual on the observed price), and it directly produces an ATT estimate rather than a shock decomposition.
  \end{itemize}
  \normalsize
\end{frame}

      {\footnotesize Estimand: ATT over a defined post-event window. Method: synthetic control on the observed price series.}
    \end{center}
    \vspace{0.1em}
  \end{minipage}}
  \end{center}

  \vspace{0.2em}
  \hd{Part 2 \emph{(open; discussed at the end)}.}

  \vspace{0.5em}
  \hd{What is new in the field.}
  \begin{itemize}\setlength\itemsep{0.1em}
    \item Closest mechanical SCM-on-price precedent: Austrian retail fuel \src{Becker, Pfeifer \& Schweikert 2021, \textit{Energy Economics}}. Field default for oil-price impact: structural VAR \src{Kilian 2009; Baumeister \& Hamilton 2019} or scenario models (Dallas Fed, OIES, Kiel, IEA).
    \item \hd{Method extension:} retail fuel $\rightarrow$ crude benchmark; cross-country $\rightarrow$ cross-asset donors; regulatory $\rightarrow$ exogenous geopolitical treatment.
    \item \hd{Two-event robustness design:} Russia 2022 used as a magnitude-validation case --- SCM counterfactual within \$$3.98$/bbl of EIA STEO's last pre-invasion structural forecast on the matched 151-day window ($\sim 5.7$ pp on implied ATT), an independent-method check that the design generalises to the Hormuz application.
  \end{itemize}
\end{frame}

  \begin{itemize}\setlength\itemsep{0.12em}

    \item \hd{SUTVA} --- qualitative donor audit on causal channels \src{Abadie 2021 \S3.1}.
    \item \hd{Inference} --- in-space placebo, 21- and 33-donor pools \src{Abadie 2010 \S3.3}; cross-event weight transfer substitutes for the low-power Hormuz placebo ($\sim$60 post-event obs).
  \end{itemize}
\end{frame}

\begin{frame}{Headline result --- ATT estimate}
  \footnotesize
  \hd{Why ATT?} One treated unit (Brent) $\rightarrow$ effect \emph{for that unit}, not a population ATE \src{Abadie 2010, 2021}. Mean gap over the post-event window --- integrates path-reversion, robust to single-day liquidity jumps.

  \vspace{0.5em}
  \begin{center}
  \begin{tabular}{@{}l c c@{}}
    \toprule
    \hd{Model}     & \hd{Russia 2022} & \hd{Hormuz 2026} \\
    \midrule
    Convex SCM         & $+29.6\%$ & $+38.6\%$ \\
    ASCM               & $+13.3\%$ & $+43.7\%$ \\
    Elastic-net        & $+21.7\%$ & $+49.8\%$ \\
    XGBoost            & $+36.8\%$ & $+36.8\%$ \\
    Bayesian Ridge     & $+0.5\%$ \smalltext{(deg.)} & $+43.6\%$ \\
    \midrule
    \hd{Ensemble median} & \textcolor{accent}{\hd{$+21.7\%$}} & \textcolor{accent}{\hd{$+43.6\%$}} \\
    IQR                  & $[+13.3,\ +29.6]$ & $[+38.6,\ +43.7]$ \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.5em}
  \hd{Reading.} Russia $+21.7\%$ within $\sim$2.5 pp of STEO cross-check; Bayesian Ridge degenerate, IQR excludes it. Hormuz $+43.6\%$ with tight IQR --- place against Dallas Fed / OIES / IEA scenarios.
  \normalsize
\end{frame}

\begin{frame}{Appendix --- donor SUTVA cleanliness}
  \footnotesize
  \hd{Permutation mean-shift test} at known $T_0$ on donor log-returns \src{Chow 1960 hypothesis + Lehmann \& Romano 2005 permutation + Phipson \& Smyth 2010 correction}. Combined flag = permutation \emph{or} Wilcoxon event-window rejects under Benjamini-Hochberg FDR ($\alpha = 0.10$).

  \vspace{0.6em}
  \begin{center}
  \begin{tabular}{@{}l r r r r@{}}
    \toprule
                & \hd{Permut.} & \hd{Event-wind.} & \hd{Combined} & \hd{Audit} \\
    \midrule
    Russia 2022 & 1 / 33       & 0 / 33           & 1 / 33        & 12 / 33    \\
    Hormuz 2026 & 0 / 33       & 1 / 33           & 1 / 33        & 0 / 33     \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.6em}
  \begin{itemize}\setlength\itemsep{0.15em}
    \item \hd{Hormuz: 32 / 33 agree.} Audit's ``no Persian-Gulf routing'' confirmed; only IronOre triggers (event-window).
    \item \hd{Russia: 20 / 33 agree.} Audit's H/M tier captures \emph{causal channels} (Wheat: Russia+Ukraine $\approx 30\%$ world exports) that daily-return tests can't reproduce at $N = 33$ after FDR. Tests' lone rejection (CNY) is China zero-COVID, not Russia-treatment.
  \end{itemize}
  \normalsize
\end{frame}
