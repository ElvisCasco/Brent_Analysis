\begin{frame}{Statistical cross-check --- shifts at $T_0$}
  \begin{itemize}\setlength\itemsep{0.3em}
    \item \hd{Permutation mean-shift} \src{Chow 1960}: split the donor's own series at $T_0$, take the
      mean gap (after $-$ before), and compare it to thousands of random before/after relabelings ---
      an unusually large gap signals a real shift. Nonparametric, no distributional assumption.
    \item \hd{HLT trend-break} \src{Harvey, Leybourne \& Taylor 2009}: tests whether the donor's own
      \emph{trend} breaks at $T_0$, valid whether the price is stationary or a random walk. Brent-free,
      so a break implicates the donor itself, not Brent's own $T_0$ move.
  \end{itemize}

  \vspace{0.4em}
  \begin{center}
  \begin{tabular}{@{}l c c@{}}
    \toprule
    \hd{Flagged at $T_0$ (of 19 retained)} & \hd{Hormuz 2026} & \hd{Russia 2022} \\
    \midrule
    HLT trend-break        & 0 / 19 & 0 / 19 \\
    Permutation mean-shift & 0 / 19 & 1 / 19 \smalltext{(CNY)} \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.4em}
  \smalltext{HLT flags nothing. The lone flag --- \hd{CNY} --- is the concurrent China zero-COVID
  reopening, not Russia-treatment, and is kept on the audit's judgment. \emph{Tests inform; the audit decides.}}
\end{frame}

\begin{frame}{Signed contamination bias}
  \footnotesize
  \hd{The critique.} If high oil \emph{crowds out demand}, activity-proxy donors are pushed \emph{down},
  dragging the synthetic down $\rightarrow$ contamination in the gap. \hd{Which way, and how big?}
  We sign each linear model's net donor-bias \smalltext{($-\sum_j w_j\,\delta_j$)} over the full post-window:

  \vspace{0.1em}
  \begin{center}
  \begin{tabular}{@{}l r r@{}}
    \toprule
    \hd{Model}     & \hd{Hormuz net bias} & \hd{Russia net bias} \\
    \midrule
    Convex SCM     & $-1.3\%$ \smalltext{(conservative)} & $+21.4\%$ \\
    ASCM           & $-2.1\%$ \smalltext{(conservative)} & $+12.7\%$ \\
    Elastic-net    & $-2.4\%$ \smalltext{(conservative)} & $+20.6\%$ \\
    Bayesian Ridge & $-7.6\%$ \smalltext{(conservative)} & $-7.1\%$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \smalltext{Linear models only (XGBoost nonlinear). Source: contamination\_bias\_decomposition.csv.}

  \vspace{0.15em}
  \begin{itemize}\setlength\itemsep{0.1em}
    \item \hd{Hormuz: bias \emph{negative}} for all linear models $\rightarrow$ demand-destruction makes the
      SCM \emph{under}-state the premium, so the headline \textcolor{accent}{\bfseries $\sim$59\%} is a \hd{lower bound}.
    \item \hd{Russia: bias \emph{positive}} $\rightarrow$ retained cyclical donors (Coffee, Platinum, SP500)
      were pushed \emph{down}, dragging the synthetic down and \emph{inflating} the gap $\rightarrow$ Russia
      is a magnitude check, not the headline.
  \end{itemize}
\end{frame}

\begin{frame}{Appendix --- in-space placebo (treated-unit inference)}
  \footnotesize
  \hd{Procedure} \src{Abadie 2010}: refit the SCM treating each donor as treated; rank Brent's post/pre RMSPE
  ratio against the $N{=}20$ units (Brent $+$ 19 donors). \hd{Permutation $p = $ rank $/\,N$}; floor $p = 1/20 = 0.05$
  $\rightarrow$ rank 1 is the strongest signal the 19-donor pool can produce.

  \vspace{0.4em}
  \begin{center}
  \begin{tabular}{@{}l c c@{}}
    \toprule
                   & \hd{Hormuz 2026} & \hd{Russia 2022} \\
    \hd{Model}     & rank\,$\rightarrow$\,$p$ & rank\,$\rightarrow$\,$p$ \\
    \midrule
    Convex SCM     & \textbf{1\,$\rightarrow$\,0.05} & 10\,$\rightarrow$\,0.50 \\
    ASCM           & \textbf{1\,$\rightarrow$\,0.05} & 12\,$\rightarrow$\,0.60 \\
    Elastic-net    & \textbf{1\,$\rightarrow$\,0.05} & 8\,$\rightarrow$\,0.40  \\
    XGBoost        & \textbf{1\,$\rightarrow$\,0.05} & 4\,$\rightarrow$\,0.20  \\
    Bayesian Ridge & \textbf{1\,$\rightarrow$\,0.05} & 12\,$\rightarrow$\,0.60 \\
    \bottomrule
  \end{tabular}
  \end{center}
  \smalltext{Source: data/validation/inference\_inspace\_brent\_summary.csv (19-donor shared pool).}

  \vspace{0.4em}
  \begin{itemize}\setlength\itemsep{0.05em}
    \item \hd{Hormuz:} Brent ranks 1 for \hd{all five models} $\rightarrow$ its break beats every placebo; strongest signal the pool allows.
    \item \hd{Russia:} mid-pack (rank 8--12) for the linear models; XGBoost is best at rank 4 ($p{=}0.20$) but still not significant --- consistent with Russia as a magnitude check, not a clean inferential case.
  \end{itemize}
\end{frame}
